cmake_minimum_required(VERSION 3.14)
project(cv_pipeline_deploy C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DEPLOY_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
set(THIRD_PARTY_ROOT "${DEPLOY_ROOT}/../3rdlibrary" CACHE PATH "Shared third-party libraries")
set(DNN_ROOT "${THIRD_PARTY_ROOT}/DNN" CACHE PATH "DNN SDK")
set(CAMERA_ROOT "${THIRD_PARTY_ROOT}/RDK_CAMERA" CACHE PATH "RDK camera SDK")
include_directories(${DEPLOY_ROOT}/include ${DNN_ROOT}/include ${CAMERA_ROOT}/include)
link_directories(${DNN_ROOT}/lib ${CAMERA_ROOT}/lib)
add_link_options(-Wl,-rpath-link,${DNN_ROOT}/lib -Wl,-rpath-link,${CAMERA_ROOT}/lib -Wl,--allow-shlib-undefined)
add_library(deploy_core STATIC src/logger.cpp src/socket_protocol.cpp src/vio_pipeline.cpp src/cleanup_stack.cpp src/sc132gs_linear_1088x1280_raw10_30fps_1lane.c)
set_target_properties(deploy_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(deploy_core PUBLIC dnn cam vpf hbmem hbrt_bayes_aarch64 cnn_intf alog pthread rt dl)
add_executable(capture_service src/capture_service.cpp)
target_link_libraries(capture_service PRIVATE deploy_core)
add_executable(inference_service src/inference_service.cpp src/raw_head_runner.cpp src/logger.cpp src/socket_protocol.cpp)
target_link_libraries(inference_service PRIVATE dnn hbmem hbrt_bayes_aarch64 cnn_intf alog pthread rt dl)
install(TARGETS capture_service inference_service RUNTIME DESTINATION bin)
